=====================================================
2. CORE BUSINESS LOGIC & BARCODE SPECIFICATIONS
=====================================================
[1. Dynamic 5-Digit Auto ItemCode Logic]
- ItemCode มีความยาว 5 หลักสม่ำเสมอ ล็อกตาม Dynamic Running Length 
- คำนวณจาก PrefixCode ในตาราง `pbitemgroup_prefix` (ความยาว 2 หรือ 3 หลัก) + เลขรันลำดับ (SeqNo)
- หาก SeqNo เต็ม MaxSeqNo (999 สำหรับ 2 หลัก, 99 สำหรับ 3 หลัก) ให้ตัด IsActive = 0 และเปิดใช้ PrefixCode ชุดใหม่

[2. Multi-Unit & Unit Conversion Override]
- มี Master Group (`pbunitgroup`) และ Master Conversion (`pbunit_conversion`)
- ในระดับ Item (`pbitem_unit`) สามารถเลือกหน่วยนับจาก Group มาใช้งาน และ **สามารถ Override อัตราแปลง (ConversionFactor) แยกตาม ItemCode ได้**

[3. Multi-Barcode & Multi-Price]
- 1 ItemCode มีได้หลายบาร์โค้ด และแต่ละบาร์โค้ดผูกกับ UnitCode ที่ต่างกันได้ (`pbitem_barcode`)
- มีระดับราคาขาย 4 ระดับ (PriceLevel1 - PriceLevel4) ประจำแต่ละบาร์โค้ด

[4. GS1 EAN-13 Internal Barcode Pattern (13 Digits)]
- รูปแบบบาร์โค้ดสากล 13 หลักพอดี:
  `2191` (Prefix 4 หลัก) + `ItemCode` (5 หลัก) + `Sequence` (3 หลัก: 000-999) + `CheckDigit` (1 หลัก Modulo 10)
  (รวม: 4 + 5 + 3 + 1 = 13 หลัก)

[5. Item Roles (Multi-Role Flags)]
- 1 ItemCode เป็นได้หลายบทบาท คุมด้วย Flags: `IsRawMaterial`, `IsFinishedGood`, `IsSalesItem`, `IsPurchaseItem`

มาตรฐานโครงสร้างตาราง:
ชื่อตารางและคอลัมน์ใช้ตัวพิมพ์เล็กทั้งหมด
ใช้ snake_case และ _ คั่นคำ
ชื่อตารางทุกตารางขึ้นต้นด้วย pb_
ชื่อตารางใช้พหูพจน์ เช่น pb_items
Primary key ใช้ id
Foreign key ใช้ <entity>_id
รหัสธุรกิจใช้ <entity>_code
ชื่อข้อมูลหลักใช้ <entity>_name
หลีกเลี่ยงชื่อกำกวม เช่น code, name, group_code เมื่อไม่ทราบบริบท
ไม่ต้องใส่ชื่อตารางนำหน้าคอลัมน์มาตรฐาน เช่น created_at
Boolean ใช้ is_, has_ หรือ can_

ใช้คำย่อเฉพาะคำมาตรฐาน เช่น id, qty, no, uom
หลีกเลี่ยงคำย่อที่อ่านยาก เช่น itm_grp_cd
ชื่อ Unique index ใช้ uq_<table>_<columns>
ชื่อ Index ใช้ idx_<table>_<columns>
ชื่อ Foreign key constraint ใช้ fk_<table>_<referenced_table>

CREATE TABLE IF NOT EXISTS pb_unit_groups (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    unit_group_code VARCHAR(30) NOT NULL,
    unit_group_name VARCHAR(150) NOT NULL,
    base_unit_id BIGINT UNSIGNED NULL,
    description VARCHAR(500) NULL,

    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_pb_unit_groups
        PRIMARY KEY (id),

    CONSTRAINT uq_pb_unit_groups_code
        UNIQUE (unit_group_code)

-- =========================================================
-- 3. หน่วยนับ
-- base_qty หมายถึงจำนวนเมื่อแปลงเป็นหน่วยฐาน
-- =========================================================

CREATE TABLE IF NOT EXISTS pb_units (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    unit_group_id BIGINT UNSIGNED NOT NULL,
    unit_code VARCHAR(30) NOT NULL,
    unit_name VARCHAR(100) NOT NULL,
    unit_symbol VARCHAR(20) NULL,
    base_qty DECIMAL(18,6) NOT NULL DEFAULT 1.000000,
    decimal_places TINYINT UNSIGNED NOT NULL DEFAULT 0,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT pk_pb_units
        PRIMARY KEY (id),

    CONSTRAINT uq_pb_units_group_code
        UNIQUE (unit_group_id, unit_code),

    CONSTRAINT fk_pb_units_unit_group
        FOREIGN KEY (unit_group_id)
        REFERENCES pb_unit_groups (id)
        ON UPDATE RESTRICT
        ON DELETE RESTRICT,

    CONSTRAINT chk_pb_units_base_qty
        CHECK (base_qty > 0),

    CONSTRAINT chk_pb_units_decimal_places
        CHECK (decimal_places <= 6)
) ENGINE=InnoDB;

CREATE INDEX idx_pb_units_unit_group_id
    ON pb_units (unit_group_id);

    CONSTRAINT pk_pb_items
        PRIMARY KEY (id),

    CONSTRAINT uq_pb_items_item_code
        UNIQUE (item_code),

    CONSTRAINT uq_pb_items_barcode
        UNIQUE (barcode),

    CONSTRAINT fk_pb_items_item_group
        FOREIGN KEY (item_group_id)
        REFERENCES pb_item_groups (id)
        ON UPDATE RESTRICT
        ON DELETE RESTRICT,

    CONSTRAINT fk_pb_items_unit_group
        FOREIGN KEY (unit_group_id)
        REFERENCES pb_unit_groups (id)
        ON UPDATE RESTRICT
        ON DELETE RESTRICT,

    CONSTRAINT fk_pb_items_base_unit
        FOREIGN KEY (base_unit_id)
        REFERENCES pb_units (id)
        ON UPDATE RESTRICT
        ON DELETE RESTRICT,

    CONSTRAINT chk_pb_items_cost_price
        CHECK (cost_price >= 0),

    CONSTRAINT chk_pb_items_selling_price
        CHECK (selling_price >= 0),

    CONSTRAINT chk_pb_items_min_stock
        CHECK (min_stock_qty >= 0),

    CONSTRAINT chk_pb_items_max_stock
        CHECK (
            max_stock_qty IS NULL
            OR max_stock_qty >= min_stock_qty

    );

  TextColumn get itemGroupCode => text().withLength(min: 1, max: 20)();
  TextColumn get itemGroupName => text().withLength(min: 1, max: 100)();
  TextColumn get parentGroupCode => text().nullable().references(Pbitemgroups, #itemGroupCode)();
  IntColumn get sortOrder => integer().withDefault(const Constant(0))();
  BoolColumn get isActive => boolean().withDefault(const Constant(true))();
  DateTimeColumn get createdAt => dateTime().withDefault(currentDateAndTime)();
  DateTimeColumn get updatedAt => dateTime().withDefault(currentDateAndTime)();

  @override
  Set<Column> get primaryKey => {itemGroupCode};
}

5. GS1 BARCODE GENERATOR UTILITY
JavaScript / Node.js:
function calculateGS1CheckDigit(payload12) {
  if (payload12.length !== 12) throw new Error('Payload must be exactly 12 digits');
  let sum = 0;
  for (let i = 0; i < 12; i++) {
    const digit = parseInt(payload12[i], 10);
    sum += (i % 2 === 0) ? digit * 1 : digit * 3;
  }
  const remainder = sum % 10;
  return remainder === 0 ? 0 : 10 - remainder;
}

function generateGS1Barcode(itemCode, seq = 0) {
  const prefix = "2191";
  const paddedItemCode = String(itemCode).padStart(5, '0');
  const paddedSeq = String(seq).padStart(3, '0');
  const payload12 = `${prefix}${paddedItemCode}${paddedSeq}`;
  return `${payload12}${calculateGS1CheckDigit(payload12)}`;
}

Dart / Flutter:
class GS1BarcodeUtil {
  static int calculateCheckDigit(String payload12) {
    if (payload12.length != 12) throw ArgumentError('Payload must be exactly 12 digits');
    int sum = 0;
    for (int i = 0; i < 12; i++) {
      int digit = int.parse(payload12[i]);
      sum += (i % 2 == 0) ? digit * 1 : digit * 3;
    }
    int remainder = sum % 10;
    return remainder == 0 ? 0 : 10 - remainder;
  }

  static String generateGS1Barcode(String itemCode, {int seq = 0}) {
    const String prefix = "2191";
    final String paddedItemCode = itemCode.padLeft(5, '0');
    final String paddedSeq = seq.toString().padLeft(3, '0');
    final String payload12 = "$prefix$paddedItemCode$paddedSeq";
    return "$payload12${calculateCheckDigit(payload12)}";
  }
}
